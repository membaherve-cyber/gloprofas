   423→async function main() {

   425→  fs.mkdirSync(CTX, { recursive: true });
   426→  if (!fs.existsSync(DONE)) fs.writeFileSync(DONE, "");
   427→
   428→  const done = new Set(
   429→    fs
   430→      .readFileSync(DONE, "utf8")
   431→      .split("\n")
   432→      .map((l) => l.trim())
   433→      .filter(Boolean)
   434→  );
   435→
   436→  const zai = await ZAI.create();
   437→  let ok = 0;
   438→  let skipped = 0;
   439→  const failed: string[] = [];
   440→
   441→  const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
   442→
   443→  for (const job of JOBS) {
   444→    if (done.has(job.file)) {
   445→      console.log(`[skip] ${job.file}`);
   446→      skipped++;
   447→      continue;
   448→    }
   449→    let success = false;
   450→    // Up to 6 attempts with rate-limit-aware exponential backoff.
   451→    for (let attempt = 1; attempt <= 6 && !success; attempt++) {
   452→      try {
   453→        const res = await zai.images.generations.create({
   454→          prompt: job.prompt,
   455→          size: job.size as "1344x768",
   456→        });
   457→        const b64 = res?.data?.[0]?.base64;
   458→        if (!b64) throw new Error("empty response");
   459→        const buf = Buffer.from(b64, "base64");
   460→        fs.writeFileSync(path.join(job.dir, job.file), buf);
   461→        for (const alias of job.also ?? []) {
   462→          fs.writeFileSync(path.join(job.dir, alias), buf);
   463→        }
   464→        console.log(
   465→          `[ok] ${job.file}${job.also?.length ? ` (+ ${job.also.join(", ")})` : ""} — ${Math.round(
   466→            buf.length / 1024
   467→          )}KB`
   468→        );
   469→        success = true;
   470→        ok++;
   471→        // Pace requests to stay under the rate limit.
   472→        await sleep(4000);
   473→      } catch (e) {
   474→        const msg = e instanceof Error ? e.message : String(e);
   475→        const isRate = msg.includes("429") || msg.toLowerCase().includes("too many");
   476→        const waitMs = isRate ? 30000 * attempt : 5000 * attempt;
   477→        console.error(
   478→          `[fail ${attempt}/6] ${job.file}: ${msg.slice(0, 120)} — waiting ${Math.round(
   479→            waitMs / 1000
   480→          )}s${isRate ? " (rate limit)" : ""}`
   481→        );
   482→        if (attempt < 6) await sleep(waitMs);
   483→      }
   484→    }
   485→    if (success) {
   486→      fs.appendFileSync(DONE, job.file + "\n");
   487→      done.add(job.file);
   488→    } else {
   489→      failed.push(job.file);
   490→    }
   491→  }
   492→
   493→  console.log(
   494→    `DONE: ${ok} generated, ${skipped} skipped, ${failed.length} failed. Failed: ${
   495→      failed.join(", ") || "none"
   496→    }`
   497→  );
   498→  if (failed.length > 0) process.exitCode = 1;

   502→  console.error("FATAL", e);
   503→  process.exit(1);
